How it runs

# runs every morning at 06:00 on our server
items = []
for page in range(1, 40):
    html = fetch(f"https://example-marketplace.no/cars?page={page}")
    for card in parse(html).select(".listing"):
        items.append(clean(card))          # price → int, km → int, dedupe by ad id
diff = compare_with_yesterday(items)       # new ads, price drops, sold
save_csv(items); push_to_google_sheet(items)
telegram_alert(diff.price_drops, chat="@client_channel")
